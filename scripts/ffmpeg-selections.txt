# Which muxers, demuxers, encoders and decoders FFmpeg is built with.
#
# Read by scripts/build-ffmpeg.py and passed to configure verbatim, after the
# --disable-* lines have switched the whole category off. Order is
# significant: a --disable-*s line must precede the --enable-* lines it scopes.
#
# Provenance: this is FFmpeg n8.1.2's set as selected by MPVKit 1.0.0, which is
# where Libavcodec, Libavutil and Libswresample used to come from, so building
# all four libraries here changed nothing about what they decode. It was
# captured by parsing FFMPEG_CONFIGURATION out of MPVKit's own build. The last
# two decoders are the external ones, dav1d for AV1 and uavs3d for AVS3; the
# script enables the libraries they wrap.
#
# To regenerate after an FFmpeg bump, build once and read the list back out of
# the artifact, which records every flag it was configured with:
#
#   python3 - <<'EOF'
#   import re, shlex
#   from pathlib import Path
#   p = "Artifacts/Libavformat.xcframework/ios-arm64/Libavformat.framework/Headers/config.h"
#   c = re.search(r'^#define FFMPEG_CONFIGURATION "(.*)"$', Path(p).read_text(), re.M)[1]
#   print("\n".join(f for f in shlex.split(c) if f.startswith((
#       "--disable-muxers", "--enable-muxer=", "--disable-demuxers", "--enable-demuxer=",
#       "--disable-encoders", "--enable-encoder=", "--disable-decoders", "--enable-decoder="))))
#   EOF
#
# Adding a codec here is not enough on its own to play it: see docs/codec-support.md.
